<p>加密部分关键是 c0 = k ^ ((k + p0) % 256) ^ c0 。k这里可以直接得到然后异或回去，c0的话，取上一个点的值异或，然后再减k就可以了。这一部分的逆向需要顺序进行，否则p0是未知的。</p>
<pre><code class='language-python' lang='python'>from PIL import Image
from decimal import *
import numpy as np
import random

getcontext().prec = 20


def f1(x):
    assert (x &gt;= 0)
    assert (x &lt;= 1)
    return 4 * x * (1 - x)


def f2(x):
    assert (x &gt;= 0)
    assert (x &lt;= 1)
    return 4 * x * (1 - x)


def f3(x):
    assert (x &gt;= 0)
    assert (x &lt;= 1)
    return 4 * x * (1 - x)


im = Image.open(&#39;flag_enc.bmp&#39;)
size = im.size
pic = np.array(im)
im.close()
r1 = Decimal(&#39;0.478706063089473894123&#39;)
r2 = Decimal(&#39;0.613494245341234672318&#39;)
r3 = Decimal(&#39;0.946365754637812381837&#39;)
w, h = size
for i in range(200):
    r1 = f1(r1)
    r2 = f2(r2)
    r3 = f3(r3)
const = 10 ** 14
p = [100, 101, 102, 103]
c = [200, 201, 202, 203]
R1, R2, R3 = r1, r2, r3
l1 = []
l2 = []
for i in range(w * h):
    l1.append(r1)
    l2.append(r2)
    r1 = f1(r1)
    r2 = f2(r2)


def decrypt(p0, c0, r1=R1, r2=R2, r3=R3):
    for x in range(w):
        for y in range(h):
            k = int(round(const * r3)) % 256
            k = bin(k)[2:].ljust(8, &#39;0&#39;)
            k = int(k[p0 % 8:] + k[:p0 % 8], 2)
            r3 = f3(r3)
            pic[y, x] = ((pic[y, x] ^ c0 ^ k) - k) % 256
            p0 = pic[y, x]
            c0 = k ^ ((k + p0) % 256) ^ c0
    t = w * h - 1
    for x in range(w - 1, -1, -1):
        for y in range(h - 1, -1, -1):
            x1 = int(round(const * l1[t])) % w
            y1 = int(round(const * l2[t])) % h
            tmp = pic[y, x]
            pic[y, x] = pic[y1, x1]
            pic[y1, x1] = tmp
            t -= 1
    return pic, size


enc_img = &#39;flag.bmp&#39;
count = 0
for p0 in p:

        im = Image.new(&#39;P&#39;, size, &#39;white&#39;)
        pixels = im.load()
        for i in range(im.size[0]):
            for j in range(im.size[1]):
                pixels[i, j] = (int(pic[j][i]))
        count += 1
        im.save(str(count) + enc_img)
